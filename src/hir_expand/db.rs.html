21#[derive(Debug, Clone, Copy, Eq, PartialEq)]
22pub enum TokenExpander<'db> {
23    /// Old-style `macro_rules` or the new macros 2.0
24    DeclarativeMacro(&'db DeclarativeMacroExpander),
25    /// Stuff like `line!` and `file!`.
26    BuiltIn(BuiltinFnLikeExpander),
27    /// Built-in eagerly expanded fn-like macros (`include!`, `concat!`, etc.)
28    BuiltInEager(EagerExpander),
29    /// `global_allocator` and such.
30    BuiltInAttr(BuiltinAttrExpander),
31    /// `derive(Copy)` and such.
32    BuiltInDerive(BuiltinDeriveExpander),
33    UnimplementedBuiltIn,
34    /// The thing we love the most here in rust-analyzer -- procedural macros.
35    ProcMacro(CustomProcMacroExpander),
36}

38#[query_group::query_group]
39pub trait ExpandDatabase: SourceDatabase {
40    #[salsa::transparent]
41    fn resolve_span(&self, span: Span) -> FileRange;
42
43    /// Fetches the expander for this macro.
44    #[salsa::transparent]
45    #[salsa::invoke(TokenExpander::macro_expander)]
46    fn macro_expander(&self, id: MacroDefId) -> TokenExpander<'_>;
47
48    /// Fetches (and compiles) the expander of this decl macro.
49    #[salsa::invoke(DeclarativeMacroExpander::expander)]
50    #[salsa::transparent]
51    fn decl_macro_expander(
52        &self,
53        def_crate: Crate,
54        id: AstId<ast::Macro>,
55    ) -> &DeclarativeMacroExpander;
56}

58fn resolve_span(db: &dyn ExpandDatabase, Span { range, anchor, ctx: _ }: Span) -> FileRange {
59    let file_id = EditionedFileId::from_span_file_id(db, anchor.file_id);
60    let anchor_offset =
61        HirFileId::from(file_id).ast_id_map(db).get_erased(anchor.ast_id).text_range().start();
62    FileRange { file_id, range: range + anchor_offset }
63}

65/// This expands the given macro call, but with different arguments. This is
66/// used for completion, where we want to see what 'would happen' if we insert a
67/// token. The `token_to_map` mapped down into the expansion, with the mapped
68/// token(s) returned with their priority.
69pub fn expand_speculative(
70    db: &dyn ExpandDatabase,
71    actual_macro_call: MacroCallId,
72    speculative_args: &SyntaxNode,
73    token_to_map: SyntaxToken,
74) -> Option<(SyntaxNode, Vec<(SyntaxToken, u8)>)> {
75    let loc = actual_macro_call.loc(db);
76    let (_, _, span) = *actual_macro_call.macro_arg_considering_derives(db, &loc.kind);
77
78    let span_map = RealSpanMap::absolute(span.anchor.file_id);
79    let span_map = SpanMap::RealSpanMap(&span_map);
80
81    // Build the subtree and token mapping for the speculative args
82    let (mut tt, undo_info) = match &loc.kind {
83        MacroCallKind::FnLike { .. } => (
84            syntax_bridge::syntax_node_to_token_tree(
85                speculative_args,
86                span_map,
87                span,
88                if loc.def.is_proc_macro() {
89                    DocCommentDesugarMode::ProcMacro
90                } else {
91                    DocCommentDesugarMode::Mbe
92                },
93            ),
94            SyntaxFixupUndoInfo::NONE,
95        ),
96        MacroCallKind::Attr { .. } if loc.def.is_attribute_derive() => (
97            syntax_bridge::syntax_node_to_token_tree(
98                speculative_args,
99                span_map,
100                span,
101                DocCommentDesugarMode::ProcMacro,
102            ),
103            SyntaxFixupUndoInfo::NONE,
104        ),
105        MacroCallKind::Derive { derive_macro_id, .. } => {
106            let MacroCallKind::Attr { censored_attr_ids: attr_ids, .. } =
107                &derive_macro_id.loc(db).kind
108            else {
109                unreachable!("`derive_macro_id` should be `MacroCallKind::Attr`");
110            };
111            attr_macro_input_to_token_tree(
112                db,
113                speculative_args,
114                span_map,
115                span,
116                true,
117                attr_ids,
118                loc.krate,
119            )
120        }
121        MacroCallKind::Attr { censored_attr_ids: attr_ids, .. } => attr_macro_input_to_token_tree(
122            db,
123            speculative_args,
124            span_map,
125            span,
126            false,
127            attr_ids,
128            loc.krate,
129        ),
130    };
131
132    let attr_arg = match &loc.kind {
133        MacroCallKind::Attr { censored_attr_ids: attr_ids, .. } => {
134            if loc.def.is_attribute_derive() {
135                // for pseudo-derive expansion we actually pass the attribute itself only
136                ast::Attr::cast(speculative_args.clone())
137                    .and_then(|attr| {
138                        if let ast::Meta::TokenTreeMeta(meta) = attr.meta()? {
139                            meta.token_tree()
140                        } else {
141                            None
142                        }
143                    })
144                    .map(|token_tree| {
145                        let mut tree = syntax_node_to_token_tree(
146                            token_tree.syntax(),
147                            span_map,
148                            span,
149                            DocCommentDesugarMode::ProcMacro,
150                        );
151                        tree.set_top_subtree_delimiter_kind(tt::DelimiterKind::Invisible);
152                        tree.set_top_subtree_delimiter_span(tt::DelimSpan::from_single(span));
153                        tree
154                    })
155            } else {
156                // Attributes may have an input token tree, build the subtree and map for this as well
157                // then try finding a token id for our token if it is inside this input subtree.
158                let item = ast::Item::cast(speculative_args.clone())?;
159                let (_, meta) =
160                    attr_ids.invoc_attr().find_attr_range_with_source_opt(db, loc.krate, &item)?;
161                if let ast::Meta::TokenTreeMeta(meta) = meta
162                    && let Some(tt) = meta.token_tree()
163                {
164                    let mut attr_arg = syntax_bridge::syntax_node_to_token_tree(
165                        tt.syntax(),
166                        span_map,
167                        span,
168                        DocCommentDesugarMode::ProcMacro,
169                    );
170                    attr_arg.set_top_subtree_delimiter_kind(tt::DelimiterKind::Invisible);
171                    Some(attr_arg)
172                } else {
173                    None
174                }
175            }
176        }
177        _ => None,
178    };
179
180    // Do the actual expansion, we need to directly expand the proc macro due to the attribute args
181    // Otherwise the expand query will fetch the non speculative attribute args and pass those instead.
182    let mut speculative_expansion = match loc.def.kind {
183        MacroDefKind::ProcMacro(ast, expander, _) => {
184            let span = crate::proc_macro_span(db, ast);
185            tt.set_top_subtree_delimiter_kind(tt::DelimiterKind::Invisible);
186            tt.set_top_subtree_delimiter_span(tt::DelimSpan::from_single(span));
187            expander.expand(
188                db,
189                loc.def.krate,
190                loc.krate,
191                &tt,
192                attr_arg.as_ref(),
193                span_with_def_site_ctxt(db, span, actual_macro_call.into(), loc.def.edition),
194                span_with_call_site_ctxt(db, span, actual_macro_call.into(), loc.def.edition),
195                span_with_mixed_site_ctxt(db, span, actual_macro_call.into(), loc.def.edition),
196            )
197        }
198        MacroDefKind::BuiltInAttr(_, it) if it.is_derive() => {
199            pseudo_derive_attr_expansion(&tt, attr_arg.as_ref()?, span)
200        }
201        MacroDefKind::Declarative(it, _) => db
202            .decl_macro_expander(loc.krate, it)
203            .expand_unhygienic(db, &tt, loc.kind.call_style(), span),
204        MacroDefKind::BuiltIn(_, it) => {
205            it.expand(db, actual_macro_call, &tt, span).map_err(Into::into)
206        }
207        MacroDefKind::BuiltInDerive(_, it) => {
208            it.expand(db, actual_macro_call, &tt, span).map_err(Into::into)
209        }
210        MacroDefKind::BuiltInEager(_, it) => {
211            it.expand(db, actual_macro_call, &tt, span).map_err(Into::into)
212        }
213        MacroDefKind::BuiltInAttr(_, it) => it.expand(db, actual_macro_call, &tt, span),
214        MacroDefKind::UnimplementedBuiltIn(_) => crate::expand_unimplemented_builtin_macro(span),
215    };
216
217    let expand_to = loc.expand_to();
218
219    fixup::reverse_fixups(&mut speculative_expansion.value, &undo_info);
220    let (node, rev_tmap) =
221        crate::token_tree_to_syntax_node(db, &speculative_expansion.value, expand_to);
222
223    let syntax_node = node.syntax_node();
224    let token = rev_tmap
225        .ranges_with_span(span_map.span_for_range(token_to_map.text_range()))
226        .filter_map(|(range, ctx)| syntax_node.covering_element(range).into_token().zip(Some(ctx)))
227        .map(|(t, ctx)| {
228            // prefer tokens of the same kind and text, as well as non opaque marked ones
229            // Note the inversion of the score here, as we want to prefer the first token in case
230            // of all tokens having the same score
231            let ranking = ctx.is_opaque(db) as u8
232                + 2 * (t.kind() != token_to_map.kind()) as u8
233                + 4 * ((t.text() != token_to_map.text()) as u8);
234            (t, ranking)
235        })
236        .collect();
237    Some((node.syntax_node(), token))
238}

240impl<'db> TokenExpander<'db> {
241    fn macro_expander(db: &'db dyn ExpandDatabase, id: MacroDefId) -> TokenExpander<'db> {
242        match id.kind {
243            MacroDefKind::Declarative(ast_id, _) => {
244                TokenExpander::DeclarativeMacro(db.decl_macro_expander(id.krate, ast_id))
245            }
246            MacroDefKind::BuiltIn(_, expander) => TokenExpander::BuiltIn(expander),
247            MacroDefKind::BuiltInAttr(_, expander) => TokenExpander::BuiltInAttr(expander),
248            MacroDefKind::BuiltInDerive(_, expander) => TokenExpander::BuiltInDerive(expander),
249            MacroDefKind::BuiltInEager(_, expander) => TokenExpander::BuiltInEager(expander),
250            MacroDefKind::ProcMacro(_, expander, _) => TokenExpander::ProcMacro(expander),
251            MacroDefKind::UnimplementedBuiltIn(_) => TokenExpander::UnimplementedBuiltIn,
252        }
253    }
